TEST_GENERATION_PROMPT = """
Ты — педагогический дизайнер. Создай {n} тестовых вопросов по материалу.

Уровни по таксономии Блума:
  - Запоминание (20%): воспроизведение фактов
  - Понимание (30%): объяснение своими словами
  - Применение (30%): решение задач
  - Анализ (20%): сравнение, выводы

Материал урока:
{lesson_content}

Для каждого вопроса:
  - Вопрос (один правильный ответ или открытый)
  - 4 варианта ответа (если закрытый)
  - Правильный ответ + объяснение почему
  - Подсказка если студент застрял
  - Тег сложности: easy/medium/hard

Отвечай в JSON: [{{"question": ..., "answers": [...], "correct": ..., ...}}]
"""

# Адаптация сложности:
def get_next_question(student: Student, topic: str) -> Question:
    accuracy = student.get_accuracy(topic, last_n=5)

    if accuracy > 0.85:
        difficulty = "hard"    # хорошо справляется → сложнее
    elif accuracy < 0.60:
        difficulty = "easy"    # плохо → проще + объяснение
    else:
        difficulty = "medium"

    return question_bank.get(topic=topic, difficulty=difficulty)

## Чат-помощник по материалу урока

```python
# Студент задаёт вопрос по уроку → RAG по контенту курса

class CourseAssistant:
    def __init__(self, course_db: VectorDB):
        self.course_db = course_db

    def answer(self, student_question: str, lesson_id: str) -> str:
        # 1. Найти релевантные части урока
        relevant_chunks = self.course_db.search(
            query=student_question,
            filter={"lesson_id": lesson_id},
            top_k=3
        )

        # 2. Claude отвечает со ссылкой на материал
        return llm.chat(
            system="Ты — ассистент курса. Объясняй на основе материала урока.",
            context=relevant_chunks,
            question=student_question,
            instruction="Если вопрос не по теме урока — скажи об этом"
        )
```
